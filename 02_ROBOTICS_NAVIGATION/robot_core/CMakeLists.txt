cmake_minimum_required(VERSION 3.20)
project(robot_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(robot_core src/world.cpp src/kinematics.cpp src/lidar.cpp src/estimation.cpp src/planning.cpp src/control.cpp)
target_include_directories(robot_core PUBLIC include)
target_compile_options(robot_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(robot_demo src/demo.cpp)
target_link_libraries(robot_demo PRIVATE robot_core)
add_executable(robot_tests tests/test_robot.cpp)
target_link_libraries(robot_tests PRIVATE robot_core)
enable_testing()
add_test(NAME robot_behavior COMMAND robot_tests)
add_test(NAME collision_free_mission COMMAND robot_demo ${CMAKE_CURRENT_BINARY_DIR}/trajectory.csv)
