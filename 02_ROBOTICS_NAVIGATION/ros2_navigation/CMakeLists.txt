cmake_minimum_required(VERSION 3.20)
project(portfolio_navigation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_msgs REQUIRED)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(../robot_core robot_core)
add_executable(portfolio_sim src/simulator.cpp)
target_link_libraries(portfolio_sim robot_core)
ament_target_dependencies(portfolio_sim rclcpp geometry_msgs sensor_msgs nav_msgs tf2_ros)
target_compile_options(portfolio_sim PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS portfolio_sim DESTINATION lib/${PROJECT_NAME})
add_executable(velocity_guard src/velocity_guard.cpp)
ament_target_dependencies(velocity_guard rclcpp geometry_msgs sensor_msgs tf2_msgs)
target_compile_options(velocity_guard PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS velocity_guard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY worlds DESTINATION share/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
